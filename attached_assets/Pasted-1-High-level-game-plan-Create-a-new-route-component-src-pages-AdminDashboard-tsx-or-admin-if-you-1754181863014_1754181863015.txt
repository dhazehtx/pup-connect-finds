1. 📄  Create src/pages/AdminDashboard.tsx
------------------------------------------------
import { useEffect, useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { supabase } from '@/supabaseClient';

export default function AdminDashboard() {
  const nav = useNavigate();
  const [metrics, setMetrics] = useState(null);

  useEffect(() => {
    const session = supabase.auth.getSession();
    const profile = session?.user?.user_metadata;
    if (!profile?.is_admin) {
      nav('/home');        // block non-admins
      return;
    }

    (async () => {
      const [{ count: userCnt }, { count: listingCnt }, { count: reportCnt }] = await Promise.all([
        supabase.from('profiles').select('*', { count: 'exact', head: true }),
        supabase.from('listings').eq('status', 'active').select('*', { count: 'exact', head: true }),
        supabase.from('reports').eq('status', 'open').select('*', { count: 'exact', head: true }),
      ]);
      setMetrics({ userCnt, listingCnt, reportCnt });
    })();
  }, []);

  if (!metrics) return <p className="p-8">Loading…</p>;

  return (
    <main className="p-8 space-y-6">
      <h1 className="text-2xl font-bold">🐾 My Pup Admin Dashboard</h1>

      <section className="grid grid-cols-1 sm:grid-cols-3 gap-4">
        <StatCard label="Total Users" value={metrics.userCnt} />
        <StatCard label="Active Listings" value={metrics.listingCnt} />
        <StatCard label="Open Reports" value={metrics.reportCnt} />
      </section>
    </main>
  );
}

function StatCard({ label, value }) {
  return (
    <div className="rounded-xl border p-4 shadow-md">
      <p className="text-gray-500">{label}</p>
      <p className="text-3xl font-semibold">{value ?? 0}</p>
    </div>
  );
}

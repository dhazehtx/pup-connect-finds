1 Mirror Stripe products in Supabase
You already inserted rows manually, but automation keeps them fresh:

ts
Copy
Edit
// scripts/syncStripeToSupabase.ts
import Stripe from 'stripe';
import { createClient } from '@supabase/supabase-js';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY as string);
const supabase = createClient(process.env.SUPABASE_URL!, process.env.SUPABASE_SERVICE_ROLE!);

(async () => {
  const products = await stripe.products.list({ limit: 100 });
  for (const p of products.data) {
    const price = (await stripe.prices.list({ product: p.id })).data[0];

    await supabase
      .from('products')
      .upsert({
        id: p.id,                        // prod_…
        name: p.name,
        price_id: price.id,              // price_…
        price: price.unit_amount!,       // cents
        currency: price.currency,
        image_url: p.images[0] ?? null,  // you can push your own later
        is_active: !p.metadata.hidden,
      });
  }
})();
Run locally or as a GitHub/Replit cron whenever you add products in Stripe.

3 Client-side: fetch and display

// pages/marketplace/index.tsx
import { useEffect, useState } from 'react';
import { supabase } from '@/lib/supabase';
import ProductCard from '@/components/ProductCard';

export default function StorePage() {
  const [products, setProducts] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    supabase
      .from('products')
      .select('*')
      .eq('is_active', true)
      .order('price', { ascending: true })
      .then(({ data, error }) => {
        if (error) console.error(error);
        setProducts(data || []);
        setLoading(false);
      });
  }, []);

  if (loading) return <p className="text-center mt-10">Loading…</p>;

  return (
    <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6 p-4">
      {products.map((p) => (
        <ProductCard key={p.id} product={p} />
      ))}
    </div>
  );
}

5 Buy button (Stripe redirect)

// components/BuyButton.tsx
import { useState } from 'react';
import axios from 'axios';

export default function BuyButton({ priceId }: { priceId: string }) {
  const [loading, setLoading] = useState(false);

  const handleClick = async () => {
    setLoading(true);
    try {
      const { data } = await axios.post('/api/create-checkout-session', {
        price_id: priceId,
        quantity: 1,
        user_id: localStorage.getItem('uid'), // or pull from auth context
      });
      window.location.href = data.url;       // Stripe Checkout
    } catch (err) {
      console.error(err);
      alert('Payment error');
    } finally {
      setLoading(false);
    }
  };

  return (
    <button
      onClick={handleClick}
      disabled={loading}
      className="bg-primaryBlue text-white rounded-lg py-2 hover:bg-primaryBlue/90 disabled:opacity-50"
    >
      {loading ? 'Redirecting…' : 'Buy now'}
    </button>
  );
}
Client key: you don’t need the Stripe publishable key here because you’re redirecting—Stripe-JS is loaded within Checkout.
If you ever switch to Payment Element or Link, add loadStripe(PUBLISHABLE_KEY).

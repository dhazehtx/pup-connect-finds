// create in client/src/lib/fetchWithRetry.ts
export async function fetchWithRetry(
  url: string,
  opts: RequestInit = {},
  { retries = 2, timeoutMs = 8000 } = {}
) {
  for (let attempt = 0; attempt <= retries; attempt++) {
    const ctrl = new AbortController();
    const t = setTimeout(() => ctrl.abort(), timeoutMs);
    try {
      const res = await fetch(url, { ...opts, signal: ctrl.signal });
      clearTimeout(t);
      if (!res.ok) {
        const text = await res.text().catch(() => "");
        throw new Error(`HTTP ${res.status}: ${text || res.statusText}`);
      }
      return await res.json();
    } catch (err) {
      clearTimeout(t);
      if (attempt === retries) throw err;
      await new Promise(r => setTimeout(r, 500 * 2 ** attempt));
    }
  }
}

// in ServicesTab.tsx (or useServices hook)
import { useEffect, useState } from "react";
import { fetchWithRetry } from "@/lib/fetchWithRetry";

type Service = {
  id: string;
  name: string;
  type: string;
  location?: string;
  is_active?: boolean;
};

export default function ServicesTab() {
  const [data, setData] = useState<Service[] | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  async function load() {
    setLoading(true);
    setError(null);
    try {
      const json = await fetchWithRetry("/api/services?limit=24");
      // Accept empty array as a valid state
      if (Array.isArray(json)) setData(json);
      else if (Array.isArray(json?.data)) setData(json.data);
      else setData([]);
    } catch (e: any) {
      console.error("Services load failed:", e);
      setError(e?.message || "Unknown error");
      setData([]);
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => { load(); }, []);

  if (loading) return <div className="p-6 text-slate-600">Loading services…</div>;
  if (error) {
    return (
      <div className="p-6 text-center">
        <p className="mb-3 text-red-600">Failed to load services.</p>
        <p className="mb-4 text-xs text-slate-500">Details: {error}</p>
        <button onClick={load} className="px-4 py-2 rounded-xl bg-blue-600 text-white">Retry</button>
      </div>
    );
  }
  if (!data || data.length === 0) {
    return <div className="p-6 text-slate-600">No services yet. Try adjusting filters.</div>;
  }

  // …existing list/grid render
}

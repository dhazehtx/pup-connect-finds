// ✅ RESET FILE CONTENT BELOW THIS LINE
import { useEffect, useState } from "react";
import { supabase } from "../utils/supabaseClient";
import ListingCard from "../components/ListingCard";
import SkeletonGrid from "../components/SkeletonGrid";
import EmptyState from "../components/EmptyState";

interface Listing {
  id: string;
  name: string;
  breed: string;
  price: number;
  images: string[];
  created_at: string;
  // ➕ add any extra columns you need here
}

export default function Explore() {
  const [listings, setListings] = useState<Listing[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchListings() {
      const { data, error } = await supabase
        .from("listings")
        .select("*")
        .eq("status", "active")
        .order("created_at", { ascending: false });

      if (error) console.error("Supabase error:", error);
      setListings(data ?? []);
      setLoading(false);
    }

    fetchListings();
  }, []);

  return (
    <div className="container mx-auto px-4 py-8">
      {/* Header & Advanced Filters (leave as‑is) */}
      <header className="mb-8">
        {/* existing Search & Filter component import here */}
      </header>

      {/* Listings grid */}
      {loading ? (
        <SkeletonGrid count={6} />
      ) : listings.length === 0 ? (
        <EmptyState message="No puppies found. Try adjusting your filters." />
      ) : (
        <div className="grid gap-6 sm:grid-cols-2 md:grid-cols-3 xl:grid-cols-4">
          {listings.map(listing => (
            <ListingCard key={listing.id} listing={listing} />
          ))}
        </div>
      )}
    </div>
  );
}

export default function SkeletonGrid({ count = 6 }) {
  return (
    <div className="grid gap-6 sm:grid-cols-2 md:grid-cols-3 xl:grid-cols-4 animate-pulse">
      {Array.from({ length: count }).map((_, i) => (
        <div key={i} className="h-64 rounded-xl bg-gray-200" />
      ))}
    </div>
  );
}

2) Server: notification enqueue helper

// lib/notify.ts
import { supabaseAdmin } from "@/lib/supabaseAdmin";
import { DateTime } from "luxon";

type NType =
  | 'like' | 'comment' | 'follow' | 'message'
  | 'order_paid' | 'order_refund'
  | 'provider_app_submitted' | 'provider_app_approved' | 'provider_app_rejected';

type NotifyArgs = {
  recipientId: string;
  actorId?: string | null;
  type: NType;
  entityTable?: string | null;
  entityId?: string | null;
  message: string;
  meta?: Record<string, any>;
  targetUrl?: string;     // absolute route like '/post/123'
};

const THROTTLE_MAX_PER_10M = 20;

function bucket15m(): string {
  const dt = DateTime.now().toUTC();
  const b = dt.set({ minute: Math.floor(dt.minute / 15) * 15, second: 0, millisecond: 0 });
  return b.toISO();
}

export async function createNotification(args: NotifyArgs) {
  // 1) Throttle per (actor->recipient) in last 10m to avoid spam
  if (args.actorId) {
    const { data: countRows } = await supabaseAdmin
      .from('notifications')
      .select('id', { count: 'exact', head: true })
      .eq('recipient_id', args.recipientId)
      .eq('actor_id', args.actorId)
      .gte('created_at', DateTime.utc().minus({ minutes: 10 }).toISO());

    if ((countRows?.length ?? 0) > THROTTLE_MAX_PER_10M) {
      return { throttled: true };
    }
  }

  // 2) Respect recipient preferences (if row absent, assume true)
  const { data: pref } = await supabaseAdmin
    .from('notification_preferences')
    .select('*')
    .eq('user_id', args.recipientId)
    .maybeSingle();
  const typeKey =
    args.type === 'like' ? 'likes'
    : args.type === 'comment' ? 'comments'
    : args.type === 'follow' ? 'follows'
    : args.type === 'message' ? 'messages'
    : args.type.startsWith('order_') ? 'order_updates'
    : 'provider_app';
  if (pref && pref[typeKey] === false) return { suppressed: true };

  // 3) Burst grouping within 15m for same (recipient, type, entityId)
  const bucket = bucket15m();
  const base = {
    recipient_id: args.recipientId,
    actor_id: args.actorId ?? null,
    type: args.type as any,
    entity_table: args.entityTable ?? null,
    entity_id: args.entityId ?? null,
    message: args.message,
    meta: args.meta ?? {},
    target_url: args.targetUrl ?? null,
    bucket_key: `${args.recipientId}:${args.type}:${args.entityId ?? ''}:${bucket}`,
    is_read: false,
    read_at: null
  };

  // Try to find an existing row in same bucket to update message/meta (e.g., "and 3 others")
  const { data: existing } = await supabaseAdmin
    .from('notifications')
    .select('id, meta, message')
    .eq('recipient_id', args.recipientId)
    .eq('type', args.type)
    .eq('entity_id', args.entityId)
    .eq('bucket_key', base.bucket_key)
    .limit(1).maybeSingle();

  if (existing) {
    const meta = existing.meta || {};
    const actors: string[] = Array.from(new Set([...(meta.actors ?? []), args.actorId].filter(Boolean))) as string[];
    meta.actors = actors;
    const groupedMsg = actors.length > 1
      ? `${args.message} (+${actors.length - 1} others)`
      : args.message;

    await supabaseAdmin.from('notifications')
      .update({ meta, message: groupedMsg, created_at: new Date().toISOString() })
      .eq('id', existing.id);

    return { grouped: true, id: existing.id };
  }

  // Insert new
  const { data, error } = await supabaseAdmin.from('notifications').insert(base).select('id').single();
  if (error) throw error;
  return { id: data.id };
}

/** Fanout to all admins for provider app submissions */
export async function notifyAllAdminsProviderSubmission(appId: string, actorId: string, providerId: string) {
  const { data: admins } = await supabaseAdmin.from('admins').select('user_id');
  if (!admins || admins.length === 0) return;

  await Promise.all(
    admins.map(a => createNotification({
      recipientId: a.user_id,
      actorId,
      type: 'provider_app_submitted',
      entityTable: 'provider_applications',
      entityId: appId,
      message: 'New provider application submitted.',
      meta: { providerId: providerId, applicationId: appId },
      targetUrl: `/admin/inbox?app=${appId}`
    }))
  );
}

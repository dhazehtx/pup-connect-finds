1 ▸ Centralize the style in Tailwind
Add a custom component class inside tailwind.config.js:

js
Copy
Edit
// tailwind.config.js
module.exports = {
  content: [/* … */],
  theme: {
    extend: {
      colors: {
        primary: {
          600: '#0a54ff',
          700: '#0046db',
          /* … */
        },
      },
    },
  },
  plugins: [
    function ({ addComponents, theme }) {
      addComponents({
        /* primary (royal-blue) button — ALWAYS white text */
        '.btn-primary': {
          @apply inline-flex items-center justify-center rounded-2xl
                  bg-primary-600 text-white font-medium
                  hover:bg-primary-700 transition focus:outline-none;
        },
      });
    },
  ],
};
Tailwind will now generate .btn-primary with text-white baked in.

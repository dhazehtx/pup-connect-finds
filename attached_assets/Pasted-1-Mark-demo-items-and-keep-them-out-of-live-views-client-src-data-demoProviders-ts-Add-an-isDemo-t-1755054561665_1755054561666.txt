2) Centralize provider selection (guest vs live) in one hook
Create (or replace) the hook below. It never returns demo items when signed-in, and its React-Query key ensures the cache flips when auth changes.

client/src/hooks/useProviders.ts
ts
Copy
Edit
import { useEffect } from "react";
import { useQuery, useQueryClient } from "@tanstack/react-query";
import { DEMO_PROVIDERS, ServiceProvider } from "@/data/demoProviders";

// Replace this with your real auth signal (Clerk, NextAuth, Supabase, etc.)
import { useAuthStatus } from "@/lib/authStatus"; 
// ^ should return { isSignedIn: boolean }. If you don't have it,
// temporarily implement a stub that returns your real flag.

type UseProvidersResult = {
  providers: ServiceProvider[];
  isDemo: boolean;
  isLoading: boolean;
  isError: boolean;
};

export function useProviders(): UseProvidersResult {
  const { isSignedIn } = useAuthStatus(); // <- your real auth
  const queryClient = useQueryClient();

  // Flip cache when auth status changes so we don’t “reuse” demo results
  useEffect(() => {
    queryClient.invalidateQueries({ queryKey: ["providers"] });
  }, [isSignedIn, queryClient]);

  const { data, isLoading, isError } = useQuery({
    queryKey: ["providers", isSignedIn ? "live" : "guest"], // <- cache split
    queryFn: async () => {
      if (!isSignedIn) {
        // Guests see demo only
        return DEMO_PROVIDERS;
      }
      // Signed-in: fetch live only — NO fallback to demo
      const res = await fetch("/api/services/providers", {
        credentials: "include",
      });
      if (!res.ok) {
        // Live failed: show "Empty" state, not demo
        return [] as ServiceProvider[];
      }
      const json = (await res.json()) as ServiceProvider[];
      // Extra guard: if someone left demo in the DB by mistake, remove it
      return json.filter((p) => !p.isDemo);
    },
    staleTime: 0,
    gcTime: 0,
    refetchOnWindowFocus: true,
    refetchOnMount: true,
    retry: 0,
  });

  const providers = (data ?? []).filter((p) => !isSignedIn ? true : !p.isDemo);
  return {
    providers,
    isDemo: !isSignedIn,
    isLoading,
    isError,
  };
}
If you don’t have useAuthStatus, wire it to whatever you already use (Clerk’s useAuth(), NextAuth’s useSession(), etc.), returning a simple { isSignedIn: boolean }.

import { useProviders } from "@/hooks/useProviders";
import { ServiceProviderCard } from "@/components/services/ServiceProviderCard";
import { EmptyServices } from "@/components/services/EmptyServices"; // your empty state

export default function ServicesMarketplace() {
  const { providers, isDemo, isLoading, isError } = useProviders();

  // …your header / pills / search…

  if (isLoading) {
    return (
      // show your spinner/loader
      <div className="py-16 text-center text-slate-500">Loading…</div>
    );
  }

  if (isError) {
    // Live failed or guest fetch issue: show empty, not demo
    return <EmptyServices />;
  }

  if (providers.length === 0) {
    return <EmptyServices />;
  }

  return (
    <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-2 gap-6">
      {providers.map((p) => (
        <ServiceProviderCard key={p.id} provider={p} />
      ))}
    </div>
  );
}

That popup means Step 5 is missing at least one of these before calling /api/payout/start:

userId

providerId

applicationId

Let’s make this bullet-proof in two ways:

Hydrate the IDs on the client before the call (and auto-create an application if missing).

Make the server tolerant (auto-ensure provider/app if a client forgets).

You can add both; they don’t conflict.

1) Client: ensure IDs before calling Stripe

Drop this helper somewhere the Step-5 page can import.

// lib/ensureOnboardingIds.ts
export async function ensureOnboardingIds(): Promise<{
  userId: string; providerId: string; applicationId: string;
}> {
  // ask backend to resolve/ensure all IDs in one shot
  const res = await fetch("/api/onboarding/ensure-ids", { method: "POST" });
  const data = await res.json();
  if (!res.ok || !data?.success) throw new Error(data?.message || "Could not ensure onboarding IDs.");
  // persist for future steps
  try { localStorage.setItem("applicationId", data.applicationId); } catch {}
  return data as any;
}


Then wire your Step-5 button like this:

// Step5Payout.tsx
import { ensureOnboardingIds } from "@/lib/ensureOnboardingIds";
const [loading, setLoading] = useState(false);

const connectStripe = async () => {
  if (loading) return;
  setLoading(true);
  try {
    // 1) make sure we have all 3 IDs (server will create missing ones)
    const { userId, providerId, applicationId } = await ensureOnboardingIds();
    console.log("[Step5] ensured IDs", { userId, providerId, applicationId });

    // 2) start onboarding
    const res = await fetch("/api/payout/start", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ userId, providerId, applicationId, accountType }),
    });
    const data = await res.json();
    if (!res.ok || !data?.success || !data?.url) throw new Error(data?.message || "No onboarding URL.");

    window.location.href = data.url;
  } catch (e: any) {
    alert(e?.message || "Could not start Stripe onboarding.");
    console.error(e);
  } finally {
    setLoading(false);
  }
};


Make sure the button is:

<button type="button" onClick={connectStripe} className="btn btn-primary">Connect with Stripe</button>

3) Server: make /api/payout/start tolerant (optional but nice)

Allow it to create missing provider/application if the client still forgets.

// app/api/payout/start/route.ts (additions)
export const runtime = "nodejs";
export const dynamic = "force-dynamic";

function okId(x?: string | null) { return typeof x === "string" && x.length > 0; }

export async function POST(req: NextRequest) {
  try {
    const { userId, providerId: inProviderId, applicationId: inAppId, accountType } = await req.json();
    if (!okId(userId)) return NextResponse.json({ success:false, message:"Missing userId" }, { status:400 });

    const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

    // ensure provider
    let providerId = inProviderId as string | undefined;
    if (!okId(providerId)) {
      const { data: existing } = await supabase.from("providers").select("id").eq("user_id", userId).maybeSingle();
      if (existing?.id) providerId = existing.id;
      else {
        const { data, error } = await supabase.from("providers").insert({ user_id: userId }).select("id").single();
        if (error) throw error;
        providerId = data.id;
      }
    }

    // ensure application
    let applicationId = inAppId as string | undefined;
    if (!okId(applicationId)) {
      const { data: draft } = await supabase
        .from("provider_applications")
        .select("id")
        .eq("user_id", userId)
        .eq("provider_id", providerId!)
        .in("status", ["draft","in_progress"])
        .order("updated_at", { ascending:false })
        .limit(1)
        .maybeSingle();
      if (draft?.id) applicationId = draft.id;
      else {
        const { data, error } = await supabase
          .from("provider_applications")
          .insert({ user_id: userId, provider_id: providerId!, status: "draft" })
          .select("id")
          .single();
        if (error) throw error;
        applicationId = data.id;
      }
    }

    // ... proceed to create/fetch stripe account & account link as you already do,
    // and return { success:true, url }
  } catch (e:any) {
    // existing error handling
  }
}


This way the missing ID alert can never block the user—even if the client forgets to supply them, the server recovers.

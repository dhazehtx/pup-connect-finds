import React, { useMemo, useState } from "react";
import { Button } from "@/components/ui/button";

// If you have your own UI primitives, adjust imports accordingly.

type StepKey = "WELCOME" | "BASICS" | "ID" | "CHECK" | "PAYOUT" | "DETAILS" | "TERMS" | "REVIEW";

const STEPS: StepKey[] = ["WELCOME","BASICS","ID","CHECK","PAYOUT","DETAILS","TERMS","REVIEW"];

export default function ProviderOnboard() {
  const [stepIndex, setStepIndex] = useState(0);
  const step = STEPS[stepIndex];

  // global draft state kept locally for MVP; wire to API in later steps
  const [draft, setDraft] = useState({
    legalName: "",
    phone: "",
    photoUrl: "",
    serviceTypes: [] as string[],
    radiusKm: 10,
    idStatus: "pending" as "pending" | "passed" | "failed",
    livenessPassed: false,
    checkStatus: "pending" as "pending" | "passed" | "failed",
    payoutConnected: false,
    stripeAccountId: "",
    description: "",
    pricing: "",
    availability: "",
    termsAccepted: false,
  });

  const canNext = useMemo(() => {
    switch (step) {
      case "BASICS":
        return draft.legalName.trim().length > 1 && draft.phone.trim().length > 6;
      case "ID":
        return draft.idStatus === "passed" && draft.livenessPassed;
      case "CHECK":
        return draft.checkStatus === "passed";
      case "PAYOUT":
        return !!draft.payoutConnected && !!draft.stripeAccountId;
      case "DETAILS":
        return draft.description.trim().length > 10 && draft.pricing.trim().length > 0;
      case "TERMS":
        return draft.termsAccepted;
      default:
        return true;
    }
  }, [step, draft]);

  const onBack = () => setStepIndex((i) => Math.max(0, i - 1));
  const onNext = () => setStepIndex((i) => Math.min(STEPS.length - 1, i + 1));

  return (
    <div className="mx-auto max-w-xl space-y-6 p-4">
      <ProgressBar current={stepIndex} total={STEPS.length} />
      <StepTitle step={step} />

      {step === "WELCOME" && <Welcome onNext={onNext} />}
      {step === "BASICS" && (
        <Basics
          draft={draft}
          setDraft={setDraft}
        />
      )}
      {step === "ID" && (
        <IdVerify
          draft={draft}
          setDraft={setDraft}
        />
      )}
      {step === "CHECK" && (
        <BackgroundCheck
          draft={draft}
          setDraft={setDraft}
        />
      )}
      {step === "PAYOUT" && (
        <Payout
          draft={draft}
          setDraft={setDraft}
        />
      )}
      {step === "DETAILS" && (
        <Details
          draft={draft}
          setDraft={setDraft}
        />
      )}
      {step === "TERMS" && (
        <Terms
          draft={draft}
          setDraft={setDraft}
        />
      )}
      {step === "REVIEW" && (
        <Review draft={draft} />
      )}

      <div className="flex items-center justify-between pt-4">
        <Button variant="secondary" onClick={onBack} disabled={stepIndex === 0}>
          Back
        </Button>
        <Button onClick={onNext} disabled={!canNext || stepIndex === STEPS.length - 1}>
          {stepIndex === STEPS.length - 2 ? "Finish" : "Next"}
        </Button>
      </div>
    </div>
  );
}

function ProgressBar({ current, total }: { current: number; total: number }) {
  const pct = Math.round(((current + 1) / total) * 100);
  return (
    <div className="w-full">
      <div className="h-2 w-full rounded bg-gray-200">
        <div className="h-2 rounded bg-blue-500" style={{ width: `${pct}%` }} />
      </div>
      <div className="mt-1 text-right text-xs text-gray-500">{pct}%</div>
    </div>
  );
}

  legal_name text not null,

create index if not exists idx_providers_created on providers(created_at);

create table if not exists provider_verifications (
  id uuid primary key default gen_random_uuid(),
  provider_id uuid not null references providers(id) on delete cascade,
  vendor text not null,
  id_status text not null default 'pending' check (id_status in ('pending','passed','failed')),

  created_at timestamptz not null default now()

create index if not exists idx_pv_provider on provider_verifications(provider_id);

// SOL:START lib/supabase/providers
import { createClient } from "@supabase/supabase-js";

const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL!;
const supabaseAnon = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!;
export const supabase = createClient(supabaseUrl, supabaseAnon);

export async function saveProviderBasics(userId: string, data: {
  legal_name: string; phone: string; photo_url?: string;
  service_types?: string[]; radius_km?: number;
}) {
  const { data: existing, error: e1 } = await supabase
    .from("providers")
    .select("id")
    .eq("user_id", userId)
    .maybeSingle();
  if (e1) throw e1;

  if (existing) {
    const { error } = await supabase.from("providers").update({
      legal_name: data.legal_name,
      phone: data.phone,
      photo_url: data.photo_url ?? null,
      service_types: data.service_types ?? [],
      radius_km: data.radius_km ?? 10
    }).eq("id", existing.id);
    if (error) throw error;
    return existing.id;
  } else {
    const { data: ins, error } = await supabase.from("providers").insert({
      user_id: userId,
      legal_name: data.legal_name,
      phone: data.phone,
      photo_url: data.photo_url ?? null,
      service_types: data.service_types ?? [],
      radius_km: data.radius_km ?? 10
    }).select("id").single();
    if (error) throw error;
    return ins.id;
  }
}

export async function setIdStatus(providerId: string, vendor: string, status: "pending"|"passed"|"failed", livenessPassed: boolean) {
  const { error } = await supabase.from("provider_verifications").insert({
    provider_id: providerId, vendor, id_status: status, liveness_passed: livenessPassed
  });
  if (error) throw error;
}

export async function setCheckStatus(providerId: string, status: "pending"|"passed"|"failed") {
  const { error } = await supabase.from("provider_checks").insert({
    provider_id: providerId, check_status: status
  });
  if (error) throw error;
}

export async function saveStripeAccount(providerId: string, accountId: string, accountType: "individual"|"business") {
  const { error } = await supabase.from("provider_payouts").insert({
    provider_id: providerId, stripe_account_id: accountId, account_type: accountType
  });
  if (error) throw error;
}

export async function attemptAutoVerify(providerId: string) {
  // If latest ID = passed & liveness=true, and latest check = passed → providers.status='verified'
  const { data: idRows, error: e1 } = await supabase
    .from("provider_verifications")
    .select("id_status,liveness_passed,created_at")
    .eq("provider_id", providerId)
    .order("created_at", { ascending: false })
    .limit(1);
  if (e1) throw e1;

  const { data: chkRows, error: e2 } = await supabase
    .from("provider_checks")
    .select("check_status,created_at")
    .eq("provider_id", providerId)
    .order("created_at", { ascending: false })
    .limit(1);
  if (e2) throw e2;

  const idOk = idRows?.[0]?.id_status === "passed" && !!idRows?.[0]?.liveness_passed;
  const chkOk = chkRows?.[0]?.check_status === "passed";
  if (idOk && chkOk) {
    const { error } = await supabase
      .from("providers")
      .update({ status: "verified" })
      .eq("id", providerId);
    if (error) throw error;
    return "verified";
  }
  return "pending";
}

// SOL:START ProviderOnboard_patch_state
// place near other useState calls
const [providerId, setProviderId] = useState<string>("");

// simple POST helper
async function postJSON(url: string, body: any) {
  const res = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
  if (!res.ok) throw new Error(await res.text());
  return res.json();
}
// SOL:END ProviderOnboard_patch_state

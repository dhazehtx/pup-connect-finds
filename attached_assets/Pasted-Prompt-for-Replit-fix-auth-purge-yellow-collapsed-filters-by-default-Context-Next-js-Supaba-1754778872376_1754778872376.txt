In the SignIn form component:

Prevent double submit.

Await session before redirect.

Don’t redirect in a useEffect that might run before the session cookie is written.

tsx
Copy
const onSubmit = async (e: React.FormEvent) => {
  e.preventDefault();
  setLoading(true);
  const emailTrim = email.trim();
  const passwordTrim = password.trim();

  const { error } = await sb().auth.signInWithPassword({
    email: emailTrim,
    password: passwordTrim,
  });
  if (error) { setLoading(false); setError(error.message); return; }

  // Wait until Supabase reports a non-null session
  let tries = 0, session = null;
  while (!session && tries < 10) {
    const res = await sb().auth.getSession();
    session = res.data.session;
    if (!session) await new Promise(r => setTimeout(r, 100));
    tries++;
  }

  // Now redirect; middleware will see the cookie
  router.replace('/home'); // or the intended return path
};

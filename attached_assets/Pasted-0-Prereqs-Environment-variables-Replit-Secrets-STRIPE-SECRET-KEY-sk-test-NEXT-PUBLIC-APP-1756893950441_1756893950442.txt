0) Prereqs

Environment variables (Replit “Secrets”):

STRIPE_SECRET_KEY=sk_test_...
NEXT_PUBLIC_APP_URL=https://<your-app-url>      # used for return_url/refresh_url


Database (once): ensure your providers table has a column for the Stripe account id:

alter table providers add column if not exists stripe_account_id text;

1) Backend – create/fetch Connect account + Account Link

Create app/api/payout/start/route.ts (App Router). If you’re on Pages Router, use pages/api/payout/start.ts and export a default handler.

import { NextRequest, NextResponse } from "next/server";
import Stripe from "stripe";
import { createClient } from "@supabase/supabase-js";

const ORIGIN = process.env.NEXT_PUBLIC_APP_URL ?? "http://localhost:3000";
const SUPABASE_URL = process.env.NEXT_PUBLIC_SUPABASE_URL!;
const SUPABASE_SERVICE_ROLE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY!;

export async function POST(req: NextRequest) {
  try {
    const { userId, providerId, applicationId, accountType, ein } = await req.json();

    if (!userId || !providerId || !applicationId) {
      return NextResponse.json({ success: false, message: "Missing fields." }, { status: 400 });
    }

    const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, { apiVersion: "2024-06-20" as any });
    const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

    // 1) Fetch provider to see if they already have a Connect account
    const { data: provider, error: pErr } = await supabase
      .from("providers")
      .select("id, stripe_account_id")
      .eq("id", providerId)
      .single();
    if (pErr) throw pErr;

    let accountId = provider?.stripe_account_id;

    // 2) Create Connect account if none
    if (!accountId) {
      const account = await stripe.accounts.create({
        type: "express",                               // (or 'standard')
        capabilities: { transfers: { requested: true }, card_payments: { requested: true } },
        business_type: accountType === "business" ? "company" : "individual",
        // DO NOT send EIN directly here; Stripe collects it during onboarding.
        metadata: { providerId, userId, applicationId },
      });
      accountId = account.id;

      const { error: uErr } = await supabase
        .from("providers")
        .update({ stripe_account_id: accountId })
        .eq("id", providerId);
      if (uErr) throw uErr;
    }

    // 3) Create an Account Link (onboarding URL)
    const link = await stripe.accountLinks.create({
      account: accountId!,
      type: "account_onboarding",
      return_url: `${ORIGIN}/services/onboarding/details?from=stripe`,   // Step 6 page
      refresh_url: `${ORIGIN}/services/onboarding/payout?refresh=1`,
    });

    // (Optional) mark step5 as started so your stepper shows progress
    await supabase
      .from("provider_applications")
      .update({ step5_status: "started", updated_at: new Date().toISOString() })
      .eq("id", applicationId)
      .eq("user_id", userId);

    return NextResponse.json({ success: true, url: link.url });
  } catch (e: any) {
    console.error("[payout/start]", e);
    return NextResponse.json({ success: false, message: e?.message || "Internal error" }, { status: 500 });
  }
}


Notes
• Account type: we pass business_type to hint the flow (individual vs company), but Stripe will collect the right fields regardless.
• EIN field: let Stripe collect during onboarding; you can still store it locally if you want, but you don’t send it here.

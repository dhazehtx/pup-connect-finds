2) Wrap non-critical API reads with a “safe fetch” (prevents throws)

If your layout or top-level provider loads theme/notifications on mount, replace the raw fetch with this helper so failures return null instead of throwing.

// lib/safeRequest.ts
export async function safeJson<T = any>(url: string, init?: RequestInit): Promise<T | null> {
  try {
    const res = await fetch(url, { credentials: 'include', ...init });
    if (!res.ok) return null;
    return (await res.json()) as T;
  } catch {
    return null;
  }
}


Then in the places that call those endpoints:

// e.g., app/providers/ThemeProvider.tsx or layout.tsx
import { safeJson } from '@/lib/safeRequest';

useEffect(() => {
  (async () => {
    const prefs = await safeJson<{ theme?: 'light'|'dark'; color?: string }>('/api/support/preferences');
    setTheme(prefs?.theme ?? 'light');
    setAccent(prefs?.color ?? '#2563eb');
  })();
}, []);


Do the same for notifications preload:

const inbox = await safeJson('/api/notifications');
// if (inbox) setNotifications(inbox);


This keeps the console clean and stops errors from interfering with your wizard.

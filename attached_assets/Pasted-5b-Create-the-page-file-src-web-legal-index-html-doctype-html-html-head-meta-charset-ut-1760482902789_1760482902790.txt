    import { showToast } from "/lib/toast.js";

    const statePick = document.getElementById("statePick");
    const vetNote   = document.getElementById("vetNote");
    const strictEl  = document.getElementById("strictCount");
    const modEl     = document.getElementById("moderateCount");
    const lenEl     = document.getElementById("lenientCount");

    // simple demo counts; replace with real metrics later
    strictEl.textContent   = "18";
    modEl.textContent      = "23";
    lenEl.textContent      = "9";

    async function loadVet(state) {
      const r = await fetch(`/api/legal/requirements?state=${state}&category=veterinary_care`);
      if (!r.ok) { vetNote.textContent = "Could not load requirements."; return; }
      const { rule } = await r.json();
      if (!rule) {
        vetNote.innerHTML = `No specific rule stored for ${state}. In general, professional veterinary licensing is required in the U.S.`;
        return;
      }
      vetNote.innerHTML = `
        <strong>${state}</strong>: ${rule.notes || "License required."}
        <div class="muted" style="margin-top:6px">
          requires_license: ${rule.requires_license ? "yes" : "no"} •
          hard_block: ${rule.hard_block ? "yes" : "no"} •
          allowed_home_based: ${rule.allowed_home_based ? "yes" : "no"}
        </div>
      `;
    }

    statePick.addEventListener("change", () => loadVet(statePick.value));
    // default state:
    statePick.value = "TX";
    loadVet("TX");

1) Make server/index.ts only build the Express app (no listen here)
// server/index.ts
import express from 'express';

const app = express();
app.use(express.json());

// --- API routes (already created in your project) ---
import providerSubmit from './routes/provider/submit';
import adminProviders from './routes/admin/providers';
import adminQueue from './routes/admin/queue';
import adminInbox from './routes/admin/inbox';

app.use('/api/provider/submit', providerSubmit);
app.use('/api/admin/providers', adminProviders);
app.use('/api/admin/queue', adminQueue);
app.use('/api/admin', adminInbox);

// --- simple health root so webview shows something ---
app.get('/', (_req, res) => {
  res.type('html').send(`
    <main style="display:grid;place-items:center;min-height:100vh;background:linear-gradient(135deg,#b7c6ff,#e8ddff)">

      </div>
    </main>
  `);
});

export default app;

2) Make server/start.ts the only place that listens
// server/start.ts
import app from './index';

const PORT = Number(process.env.PORT) || 5000;
const HOST = '0.0.0.0';

app.listen(PORT, HOST, () => {
  console.log(`Server listening on ${HOST}:${PORT}`);
});


If you had any other server.listen(...) or app.listen(...) (e.g., in server/index.ts, server.ts, or a Fastify-looking server.listen({ host, port, reusePort })), delete those. One listener only.

3) Keep email helper safe (no node-fetch import)
// server/lib/email.ts
// (no import for node-fetch)

const RESEND_API_KEY = process.env.RESEND_API_KEY || '';
const FROM_EMAIL = process.env.FROM_EMAIL || 'My Pup <noreply@mypup.example>';

export async function sendEmail(to: string | string[], subject: string, html: string) {
  if (!RESEND_API_KEY) {
    console.warn('[email] RESEND_API_KEY not set; skipping email send.');
    return;
  }
  const resp = await fetch('https://api.resend.com/emails', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${RESEND_API_KEY}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      from: FROM_EMAIL,
      to: Array.isArray(to) ? to : [to],
      subject,
      html
    })
  });
  if (!resp.ok) {
    const txt = await resp.text();
    console.error('[email] Send failed', resp.status, txt);
    throw new Error(`Email failed: ${resp.status}`);
  }
}

4) Ensure your run script starts server/start.ts

In Replit, whatever command runs the server should execute start.ts (or compiled JS). For TypeScript with tsx:

// package.json (snippet)
{
  "scripts": {
    "dev": "tsx server/start.ts",
    "start": "node dist/server/start.js"
  }
}

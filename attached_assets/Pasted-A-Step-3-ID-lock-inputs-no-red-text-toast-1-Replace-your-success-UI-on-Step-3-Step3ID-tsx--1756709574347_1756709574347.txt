D) Step 5 (Payout) — how to go onward (“Connect with Stripe”)

This is for Stripe Connect onboarding. The button should:

Ensure the provider has a Connect account (create one if not).

Create an Account Link (onboarding URL).

Redirect the provider to Stripe.

When Stripe sends them back to your return_url, mark Step 5 complete.

1) API: create/fetch Connect account + account link
// app/api/payout/start/route.ts
import { NextRequest, NextResponse } from "next/server";
import Stripe from "stripe";
import { serverSupabase } from "@/lib/supabaseServer";

const ORIGIN = process.env.NEXT_PUBLIC_APP_URL ?? "http://localhost:3000";

export async function POST(req: NextRequest) {
  try {
    const { userId, providerId, applicationId } = await req.json();
    if (!userId || !providerId || !applicationId) {
      return NextResponse.json({ success:false, message:"Missing fields." }, { status:400 });
    }

    const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, { apiVersion: "2024-06-20" as any });
    const supabase = serverSupabase();

    // 1) get provider row that stores stripe_account_id (use your real table)
    const { data: providerRow, error: pErr } = await supabase
      .from("providers")
      .select("id, stripe_account_id")
      .eq("id", providerId)
      .single();
    if (pErr) throw pErr;

    let accountId = providerRow?.stripe_account_id;

    // 2) Create account if none
    if (!accountId) {
      const account = await stripe.accounts.create({
        type: "express",    // or "standard" if you prefer
        metadata: { providerId, userId },
      });
      accountId = account.id;
      await supabase.from("providers")
        .update({ stripe_account_id: accountId })
        .eq("id", providerId);
    }

    // 3) Create account link
    const acctLink = await stripe.accountLinks.create({
      account: accountId!,
      refresh_url: `${ORIGIN}/services/onboarding/payout?refresh=1`,
      return_url:  `${ORIGIN}/services/onboarding/details?onboarding=1`,
      type: "account_onboarding",
    });

    // Optional: mark step5 started so your stepper shows progress
    await supabase.from("provider_applications")
      .update({ step5_status: "started", updated_at: new Date().toISOString() })
      .eq("id", applicationId);

    return NextResponse.json({ success:true, url: acctLink.url });
  } catch (e:any) {
    console.error("[payout/start]", e);
    return NextResponse.json({ success:false, message: e?.message || "Internal error" }, { status:500 });
  }
}

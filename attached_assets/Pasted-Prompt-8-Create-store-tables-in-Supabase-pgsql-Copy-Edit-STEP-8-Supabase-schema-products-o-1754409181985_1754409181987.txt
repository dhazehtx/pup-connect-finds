# ✅ STEP 8 — Supabase schema: products, orders, order_items
-- Run in the SQL editor or via supabase CLI migration
-- products
create table if not exists public.products (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  stripe_price_id text,               -- nullable for manual SKUs
  image_url text,
  inventory_qty int default 0,
  is_active boolean default true,
  unit_price numeric(10,2) not null   -- keeps price handy for manual orders
);

-- orders
create table if not exists public.orders (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references auth.users(id),
  total numeric(10,2) not null,
  status text default 'pending',      -- pending | paid | shipped | cancelled
  created_at timestamp with time zone default now()
);

-- order_items
create table if not exists public.order_items (
  id serial primary key,
  order_id uuid references orders(id) on delete cascade,
  product_id uuid references products(id),
  qty int not null,
  unit_price numeric(10,2) not null
);

-- helpful index
create index if not exists idx_products_active on products(is_active);

# ✅ STEP 10 — GET products
// File: src/pages/api/products.ts  (Next.js)  OR Express /api/products
import type { NextApiRequest, NextApiResponse } from "next";
import { supabase } from "../../lib/supabaseClient";

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  const page = parseInt((req.query.page as string) || "1");
  const limit = parseInt((req.query.limit as string) || "10");
  const offset = (page - 1) * limit;
  const activeOnly = req.query.active !== "false";

  const query = supabase
    .from("products")
    .select("*", { count: "exact" })
    .range(offset, offset + limit - 1)
    .order("name");

  if (activeOnly) query.eq("is_active", true);

  const { data, error, count } = await query;
  if (error) return res.status(500).json({ error });

  res.status(200).json({ data, count, page, limit });
}

# ✅ STEP 11 — POST orders
// File: src/pages/api/orders.ts  (Next.js) OR Express /api/orders
import type { NextApiRequest, NextApiResponse } from "next";
import { supabase } from "../../lib/supabaseClient";

interface Item { product_id: string; qty: number }

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== "POST") return res.status(405).end();

  const { user_id, items }: { user_id: string; items: Item[] } = req.body;

  // fetch prices & check inventory
  const { data: products, error } = await supabase
    .from("products")
    .select("id, unit_price, inventory_qty")
    .in("id", items.map(i => i.product_id));

  if (error) return res.status(400).json({ error });

  const lineItems = items.map(i => {
    const p = products!.find(p => p.id === i.product_id)!;
    return { ...i, unit_price: p.unit_price, subtotal: p.unit_price * i.qty };
  });

  const total = lineItems.reduce((s, l) => s + l.subtotal, 0);

  // create order & items inside a transaction
  const { data: order, error: orderErr } = await supabase
    .from("orders")
    .insert({ user_id, total, status: "paid" })
    .select()
    .single();

  if (orderErr) return res.status(500).json({ orderErr });

  const orderItemsPayload = lineItems.map(li => ({
    order_id: order.id,
    product_id: li.product_id,
    qty: li.qty,
    unit_price: li.unit_price,
  }));

  const { error: itemsErr } = await supabase.from("order_items").insert(orderItemsPayload);
  if (itemsErr) return res.status(500).json({ itemsErr });

  res.status(201).json({ order, items: orderItemsPayload });
}

/* Example POST body:
{
  "user_id":"USER_UUID",
  "items":[
    {"product_id":"PRODUCT_UUID_1","qty":2},
    {"product_id":"PRODUCT_UUID_2","qty":1}
  ]
}
*/

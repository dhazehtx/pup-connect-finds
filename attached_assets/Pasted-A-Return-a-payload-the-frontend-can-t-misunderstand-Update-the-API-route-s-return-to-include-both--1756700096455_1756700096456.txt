B. Make the client handler bullet-proof (single call, clear errors, use provided URL)

Wire this to both the “Next” and “Start ID Verification” buttons and remove any other verification starters.

const [submitting, setSubmitting] = useState(false);
const [error, setError] = useState<string | null>(null);

const onStartVerification = async (e?: React.MouseEvent | React.FormEvent) => {
  e?.preventDefault?.();
  e?.stopPropagation?.();

  if (submitting) return; // prevent double fire
  setSubmitting(true);
  setError(null);         // clear stale error

  try {
    const res = await fetch("/api/verification/start", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ providerId, frontImagePath, backImagePath }),
    });

    const data = await res.json().catch(() => ({}));

    if (!res.ok || !data?.success) {
      throw new Error(data?.message || "Failed to start ID verification.");
    }

    // prefer backend-provided URL
    const dest =
      data?.url ||
      (data?.sessionId
        ? `https://verify.stripe.com/session/${data.sessionId}`
        : data?.session_id
        ? `https://verify.stripe.com/session/${data.session_id}`
        : null);

    if (!dest) throw new Error("Verification session missing from response.");

    window.location.href = dest; // go to Stripe hosted flow
  } catch (err: any) {
    setError(err?.message || "Could not start verification.");
    // show toast here if you use one; otherwise set local error state
  } finally {
    setSubmitting(false);
  }
};


Bind it:

<button onClick={onStartVerification} disabled={submitting || !frontImagePath || !backImagePath}>
  {submitting ? "Starting verification…" : "Start ID Verification"}
</button>


Also make the Next button call the same handler or navigate only after a session exists—don’t let it run a different path.

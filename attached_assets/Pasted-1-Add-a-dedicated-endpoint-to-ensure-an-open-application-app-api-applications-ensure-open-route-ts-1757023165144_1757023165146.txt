1) Add a dedicated endpoint to ensure an open application

app/api/applications/ensure-open/route.ts

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";

const SUPABASE_URL = process.env.NEXT_PUBLIC_SUPABASE_URL!;
const SR_KEY        = process.env.SUPABASE_SERVICE_ROLE_KEY!;

function fail(message: string, status = 400) {
  return NextResponse.json({ success: false, message }, { status });
}

export async function POST(req: NextRequest) {
  const sb = createClient(SUPABASE_URL, SR_KEY);

  try {
    const { userId, providerId } = await req.json();
    if (!userId)      return fail("Missing userId", 400);
    if (!providerId)  return fail("Missing providerId", 400);

    // 1) Try existing draft/in_progress
    const { data: existing, error: findErr } = await sb
      .from("provider_applications")
      .select("id")
      .eq("user_id", userId)
      .eq("provider_id", providerId)
      .in("status", ["draft","in_progress"])
      .order("updated_at", { ascending: false })
      .limit(1)
      .maybeSingle();

    if (findErr) return fail(`DB error (find application): ${findErr.message}`, 500);
    if (existing?.id) {
      return NextResponse.json({ success: true, applicationId: existing.id });
    }

    // 2) Create draft application
    const { data: created, error: insErr } = await sb
      .from("provider_applications")
      .insert({
        user_id: userId,
        provider_id: providerId,
        status: "draft",
      })
      .select("id")
      .single();

    if (insErr) return fail(`DB error (create application): ${insErr.message}`, 500);

    return NextResponse.json({ success: true, applicationId: created!.id });
  } catch (e: any) {
    console.error("[ensure-open] ERROR", e);
    return fail(e?.message || "Internal error", 500);
  }
}


Uses the Service Role key, so it bypasses RLS. If your SUPABASE_SERVICE_ROLE_KEY is missing/incorrect, you’ll get “row-level security” or similar—this endpoint will now return that message instead of {} so we can see it.

2) Update the Step-5 client to call that endpoint first
const connectStripe = async () => {
  setLoading(true);
  try {
    // You should already have userId & providerId by now
    // 1) Ensure we have an applicationId
    const a = await fetch("/api/applications/ensure-open", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ userId, providerId }),
    });
    const aData = await a.json();
    if (!a.ok || !aData?.success) throw new Error(aData?.message || "Could not prepare application.");
    const applicationId = aData.applicationId as string;

    // 2) Start Stripe onboarding
    const r = await fetch("/api/payout/start", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ userId, providerId, applicationId, accountType }),
    });
    const d = await r.json();
    if (!r.ok || !d?.success || !d?.url) throw new Error(d?.message || "Setup failed.");
    window.location.href = d.url;
  } catch (e: any) {
    console.error("[PAYOUT] connectStripe error", e);
    alert(e?.message || "Could not start Stripe onboarding.");
  } finally {
    setLoading(false);
  }
};


This surfaces the real DB error if the insert/select fails (e.g., bad env vars, missing columns, RLS).

STRIPE_SECRET_KEY

NEXT_PUBLIC_APP_URL

Table columns provider_applications:

id (uuid, default gen_random_uuid() or uuid_generate_v4())

user_id (uuid, NOT NULL)

provider_id (uuid, NOT NULL)

status (text, default 'draft')

created_at/updated_at defaults

If you’re not using the Service Role key anywhere, you need RLS policies. With Service Role, you don’t.

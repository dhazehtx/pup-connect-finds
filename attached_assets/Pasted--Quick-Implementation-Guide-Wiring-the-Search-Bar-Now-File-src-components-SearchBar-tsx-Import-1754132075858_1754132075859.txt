🚀 Quick Implementation Guide: Wiring the Search Bar Now

File: src/components/SearchBar.tsx

Import & state

import { useGlobalSearch } from "../hooks/useGlobalSearch";
const [query, setQuery] = useState("");
const results = useGlobalSearch(query);
const navigate = useNavigate();

Input element

<input
  type="text"
  className="w-full rounded-lg border px-4 py-2 focus:ring-primary-600"
  placeholder="Search puppies, breeds, or breeders…"
  value={query}
  onChange={e => setQuery(e.target.value)}
/>

Dropdown UI

{query && (
  <div className="absolute z-50 mt-1 w-full rounded-lg bg-white shadow-lg">
    {results.length === 0 ? (
      <p className="p-4 text-sm text-gray-500">No results</p>
    ) : (
      results.map(item => (
        <button
          key={item.id}
          onClick={() => {
            navigate(item.type === "listing" ? `/listing/${item.id}` : `/profile/${item.id}`);
            setQuery("");
          }}
          className="flex w-full items-center gap-3 px-4 py-2 hover:bg-gray-50"
        >
          {item.type === "listing" ? (
            <img src={item.image} alt={item.name} className="h-10 w-10 rounded object-cover" />
          ) : (
            <img src={item.avatar_url} alt={item.username} className="h-10 w-10 rounded-full object-cover" />
          )}
          <div className="text-left">
            {item.type === "listing" ? (
              <>
                <p className="font-medium">{item.name}</p>
                <p className="text-xs text-gray-500">{item.breed} · ${'{'}item.price{'}'}</p>
              </>
            ) : (
              <p className="font-medium">@{item.username}</p>
            )}
          </div>
        </button>
      ))
    )}
  </div>
)}

Accessibility + UX

Close dropdown on Esc:

useEffect(() => {
  const handler = (e: KeyboardEvent) => e.key === "Escape" && setQuery("");
  window.addEventListener("keydown", handler);
  return () => window.removeEventListener("keydown", handler);
}, []);

Debounce already handled inside useGlobalSearch.

Styling – swap raw class strings with the constants (CARD_BASE, etc.) after you lock the theme.

With these snippets the search bar will instantly surface both puppy listings and user profiles—exactly like Instagram’s combined search.

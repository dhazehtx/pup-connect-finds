3️⃣ Seeder Script

ts
Copy
Edit
// scripts/seedProducts.ts
import Stripe from 'stripe';
import { createClient } from '@supabase/supabase-js';
const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!);
const sb = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE);

async function seed() {
  const stripeProducts = await stripe.products.list();
  for (let p of stripeProducts.data) {
    const price = (await stripe.prices.list({ product: p.id })).data[0];
    await sb.from('products').upsert({ id: p.id, name: p.name, price_id: price.id, inventory_qty: 100, image_url: p.images[0], is_active: true });
  }
}
seed();

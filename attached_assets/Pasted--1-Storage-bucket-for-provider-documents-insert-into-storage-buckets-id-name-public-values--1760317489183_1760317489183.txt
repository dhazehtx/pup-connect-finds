-- 1. Storage bucket for provider documents
insert into storage.buckets (id, name, public) 
values ('provider-docs','provider-docs', false)
on conflict (id) do nothing;

-- 2. Provider documents table (license, insurance, certifications)
create table if not exists public.provider_documents (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  doc_type text not null check (doc_type in ('business_license','insurance_certificate','cert_cpr','cert_akc_trainer','other')),
  file_path text not null,
  file_name text not null,
  mime_type text not null,
  uploaded_at timestamp with time zone default now(),
  reviewed boolean default false,
  status text not null default 'pending' check (status in ('pending','approved','rejected')),
  notes text
);

-- 3. Service provider profile enrichments
alter table public.service_providers
  add column if not exists profile_photo_url text,
  add column if not exists years_experience int check (years_experience between 0 and 50),
  add column if not exists offers_cats boolean default false, -- dog is implicit; cats optional
  add column if not exists breed_restrictions text,          -- freeform notes
  add column if not exists rate_type text check (rate_type in ('hourly','per_visit','flat')),
  add column if not exists starting_price numeric(10,2) check (starting_price >= 0),
  add column if not exists min_booking_minutes int check (min_booking_minutes in (30,45,60,90,120)),
  add column if not exists cancellation_policy text check (cancellation_policy in ('flexible','moderate','strict')),
  add column if not exists travel_fee_enabled boolean default false,
  add column if not exists travel_fee_amount numeric(10,2) check (travel_fee_amount >= 0),
  add column if not exists additional_pet_fee_enabled boolean default false,
  add column if not exists additional_pet_fee_amount numeric(10,2) check (additional_pet_fee_amount >= 0),
  add column if not exists holiday_rate_enabled boolean default false,
  add column if not exists holiday_rate_multiplier numeric(5,2) check (holiday_rate_multiplier between 1 and 3),
  add column if not exists availability jsonb default '{}'::jsonb,  -- e.g., {"weekdays":["Mon","Tue"],"timeRanges":[["09:00","17:00"]]}
  add column if not exists background_check_status text default 'verified' check (background_check_status in ('verified')),
  add column if not exists verified_badge boolean default true,
  add column if not exists communication_policy text default 'in_app_only',
  add column if not exists policy_acknowledged boolean default false;

-- 4. RLS (allow owner to read/write their own docs)
alter table public.provider_documents enable row level security;

create policy "provider_reads_own_docs"
on public.provider_documents for select
using (auth.uid() = user_id);

create policy "provider_writes_own_docs"
on public.provider_documents for insert with check (auth.uid() = user_id);

create policy "provider_updates_own_docs"
on public.provider_documents for update using (auth.uid() = user_id);

-- 5. Storage policy (owner-only access; signed URLs for preview)
create policy "provider-docs owner read"
on storage.objects for select
to authenticated
using (bucket_id = 'provider-docs' and owner = auth.uid());

create policy "provider-docs owner write"
on storage.objects for insert
to authenticated
with check (bucket_id = 'provider-docs' and owner = auth.uid());

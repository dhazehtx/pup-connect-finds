2) Step 4 (Background Check) — consent checkbox, not a “Start” button
API: save consent (non-blocking)

app/api/applications/consent/route.ts

import { NextRequest, NextResponse } from "next/server";
import { serverSupabase } from "@/lib/supabaseServer";

export async function POST(req: NextRequest) {
  try {
    const { applicationId, userId, consent } = await req.json();
    if (!applicationId || !userId || typeof consent !== "boolean") {
      return NextResponse.json({ success:false, message:"Missing fields." }, { status:400 });
    }
    const supabase = serverSupabase();
    const { data, error } = await supabase
      .from("provider_applications")
      .update({ bgcheck_consent: consent })
      .eq("id", applicationId)
      .eq("user_id", userId)
      .select()
      .single();
    if (error) throw error;
    return NextResponse.json({ success:true, consent: data.bgcheck_consent });
  } catch (e:any) {
    console.error("[applications/consent]", e);
    return NextResponse.json({ success:false, message:e?.message || "Internal error" }, { status:500 });
  }
}

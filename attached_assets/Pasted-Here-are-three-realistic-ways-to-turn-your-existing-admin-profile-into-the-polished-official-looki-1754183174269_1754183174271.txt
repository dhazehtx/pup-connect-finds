// Prompt name: SeedCompanyProfile

/*
Goal: Automatically seed / update the official My Pup company admin profile.
Stack: TypeScript, Supabase client.
Prereqs:
  1. Place your logo in static/my-pup-logo.png
  2. Place your cover image in static/my-pup-cover.jpg
  3. Make sure you have a SERVICE_ROLE_KEY in env.
*/

import { createClient } from '@supabase/supabase-js';
import fs from 'fs';

const supabase = createClient(
  process.env.SUPABASE_URL!,
  process.env.SERVICE_ROLE_KEY!          // service role so we can update any row
);

// 1. Upload assets to storage
async function uploadAsset(path, bucket, dest) {
  const file = fs.readFileSync(path);
  await supabase.storage.from(bucket).upload(dest, file, { upsert: true, contentType: 'image/png' });
  const { publicURL } = supabase.storage.from(bucket).getPublicUrl(dest).data;
  return publicURL!;
}

const avatarUrl = await uploadAsset('static/my-pup-logo.png', 'avatars', 'my-pup-logo.png');
const coverUrl  = await uploadAsset('static/my-pup-cover.jpg', 'covers',  'my-pup-cover.jpg');

// 2. Patch the profile row (replace ID with your own)
await supabase.from('profiles').update({
  name: 'My Pup',
  username: '@MyPupOfficial',
  bio: 'Building a safe, fun & verified marketplace for dog lovers 🐾',
  avatar_url: avatarUrl,
  cover_url: coverUrl,
  is_verified: true,
  is_admin: true
}).eq('id', '8b7adf6a-eb74-43…');

console.log('✅ Company profile seeded!');
Pros: one-click repeatable in any environment.
Cons: you still need to run it (e.g., as part of a deploy script).

3️⃣ Build a “Company Settings” page (fully self-service)
New route/component /company-settings gated to is_admin === true.

Simple form with fields for name, username, bio, avatar, banner.

On submit, call supabase.from('profiles').update(…).

For uploads, use Supabase Storage + FileInput.

Save & preview—future admins can update branding in-app without database access.

Pros: non-technical teammates can handle updates.
Cons: a little extra dev work today.

Extra polish tips
Verified badge: Already using is_verified; just ensure your UI shows the blue tick when that flag is true.

Custom URL slug: If you support vanity URLs (/p/MyPup), insert or update a row in your slugs table to map mypup → profile ID.

Read-only lockout: To prevent accidental edits, conditionally hide the “Edit Profile” button when is_admin && name === 'My Pup'.

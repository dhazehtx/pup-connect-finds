Fix Store “Failed to load products” on Marketplace
Goal: Reliable products API + safe UI. No crash on empty catalog, clear errors only on real failures, and show an empty state.

import { NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
);

export async function GET() {
  try {
    // Only active products that are linked to Stripe
    const { data, error } = await supabase
      .from("products")
      .select("id,name,price_id,image_url,price,inventory_qty,is_active")
      .eq("is_active", true)
      .order("created_at", { ascending: false });

    if (error) {
      console.error("[store/products] supabase error", error);
      return NextResponse.json({ error: "DB_ERROR" }, { status: 500 });
    }

    // Never treat empty as an error
    return NextResponse.json({ items: data ?? [] }, { status: 200 });
  } catch (e) {
    console.error("[store/products] fatal", e);
    return NextResponse.json({ error: "SERVER_ERROR" }, { status: 500 });
  }
}
If you don’t use a products table yet, create it (id uuid pk, name text, price_id text, image_url text, price int, inventory_qty int default 0, is_active boolean default true). Each row should map to a Stripe Price via price_id.

"use client";
import { useEffect, useState } from "react";

type Item = {
  id: string;
  name: string;
  price_id: string | null;
  image_url: string | null;
  price: number | null;
  inventory_qty: number | null;
};

export default function StoreGrid() {
  const [items, setItems] = useState<Item[]>([]);
  const [loading, setLoading] = useState(true);
  const [errorMsg, setErrorMsg] = useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;

    const fetchProducts = async (attempt = 1) => {
      try {
        const res = await fetch("/api/store/products", { cache: "no-store" });
        if (!res.ok) throw new Error(`HTTP_${res.status}`);
        const json = await res.json();
        if (!cancelled) {
          setItems(Array.isArray(json.items) ? json.items : []);
          setErrorMsg(null);
          setLoading(false);
        }
      } catch (err) {
        console.warn("fetch products failed", err);
        if (attempt < 3) {
          // small backoff
          setTimeout(() => fetchProducts(attempt + 1), attempt * 250);
        } else if (!cancelled) {
          setErrorMsg("We couldn't load products right now.");
          setLoading(false);
        }
      }
    };

    fetchProducts();
    return () => { cancelled = true; };
  }, []);

  if (loading) {
    return <div className="p-6 text-sm text-slate-500">Loading products…</div>;
  }

  if (errorMsg) {
    return (
      <div className="m-4 rounded-xl border border-red-200 bg-red-50 p-4 text-sm text-red-800">
        {errorMsg} Please try again in a moment.
      </div>
    );
  }

  if (!items.length) {
    return (
      <div className="m-6 rounded-2xl border border-slate-200 bg-white p-8 text-center">
        <p className="text-slate-700 font-medium">No products yet</p>
        <p className="mt-1 text-slate-500 text-sm">
          Add items to your Stripe catalog and sync them to Supabase.
        </p>
      </div>
    );
  }

  return (
    <div className="grid grid-cols-1 gap-5 md:grid-cols-2">
      {items.map((p) => (
        <div key={p.id} className="rounded-2xl border border-slate-200 bg-white p-4 shadow-sm">
          {/* image */}
          <div className="aspect-[4/3] overflow-hidden rounded-xl bg-slate-100">
            {p.image_url && <img src={p.image_url} alt={p.name} className="h-full w-full object-cover" />}
          </div>
          {/* content */}
          <div className="mt-3">
            <h3 className="text-base font-semibold text-slate-900">{p.name}</h3>
            <div className="mt-1 text-primary font-bold">
              {typeof p.price === "number" ? `$${(p.price / 100).toFixed(2)}` : "—"}
            </div>
            <div className="mt-3 grid grid-cols-2 gap-3">
              <button className="btn btn-outline">Add to Cart</button>
              <button className="btn btn-primary">Buy Now</button>
            </div>
          </div>
        </div>
      ))}
    </div>
  );
}

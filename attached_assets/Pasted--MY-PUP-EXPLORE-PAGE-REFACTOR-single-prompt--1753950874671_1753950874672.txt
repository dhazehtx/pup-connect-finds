    import { supabase } from '@/lib/supabaseClient';
    import DemoExplore   from '@/components/DemoExplore';
    import PillFilterBar from '@/components/PillFilterBar';
    import PuppyGrid     from '@/components/PuppyGrid';
    import { useState, useEffect } from 'react';

    export default function ExplorePage() {
      const session = supabase.auth.session();
      if (!session) return <DemoExplore />;   // ── GUEST PATH

      /* SIGNED-IN PATH */
      const [filters, setFilters] = useState({});
      const [puppies, setPuppies] = useState([]);

      useEffect(() => {
        let q = supabase.from('puppies').select('*');
        if (filters.breed)  q = q.eq('breed', filters.breed);
        if (filters.gender) q = q.eq('gender', filters.gender);
        if (filters.price)  q = q.lte('price', filters.price);
        if (filters.age)    q = q.lt('age_months', filters.age);
        if (filters.location) q = q.ilike('location', `%${filters.location}%`);
        q.then(({ data }) => setPuppies(data));
      }, [filters]);

      return (
        <>
          <PillFilterBar filters={filters} setFilters={setFilters} />
          <PuppyGrid data={puppies} />
        </>
      );
    }
